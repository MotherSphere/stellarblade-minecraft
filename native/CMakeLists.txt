cmake_minimum_required(VERSION 3.20)
project(stellar_input LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  # Reproducible Windows binaries: no timestamps in objects or the PE header, so a commit's DLL hash can be rebuilt.
  add_compile_options(/Brepro)
  add_link_options(/Brepro)
endif()
include(CTest)
add_executable(stellar_input_tests tests/input_test.cpp)
add_test(NAME input_core COMMAND stellar_input_tests WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..)
add_executable(stellar_frame_bus_tests tests/frame_bus_test.cpp src/frame_bus.cpp)
add_test(NAME frame_bus COMMAND stellar_frame_bus_tests)
add_executable(stellar_raw_capture_tests tests/raw_capture_test.cpp)
add_test(NAME raw_capture_guard COMMAND stellar_raw_capture_tests)
add_executable(stellar_pose_stream_tests tests/pose_stream_test.cpp src/pose_stream.cpp)
add_test(NAME pose_stream COMMAND stellar_pose_stream_tests WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..)
foreach(target stellar_input_tests stellar_frame_bus_tests stellar_raw_capture_tests stellar_pose_stream_tests)
  target_include_directories(${target} PRIVATE include)
endforeach()
if(WIN32)
  add_library(stellar_input SHARED src/proxy.cpp src/raw_capture.cpp src/frame_bus.cpp src/pose_stream.cpp src/xinput.def)
  target_include_directories(stellar_input PRIVATE include)
  set_target_properties(stellar_input PROPERTIES PREFIX "" OUTPUT_NAME "xinput1_3")
  target_link_libraries(stellar_input PRIVATE bcrypt user32)
  option(STELLAR_INPUT_BUILD_FIXTURE "Build the isolated fake-host adapter (never deploy it)" OFF)
  if(STELLAR_INPUT_BUILD_FIXTURE)
    add_library(stellar_input_fixture SHARED src/proxy.cpp src/raw_capture.cpp src/frame_bus.cpp src/pose_stream.cpp src/xinput.def)
    target_include_directories(stellar_input_fixture PRIVATE include)
    target_compile_definitions(stellar_input_fixture PRIVATE STELLAR_INPUT_TEST_HOST=1)
    target_link_libraries(stellar_input_fixture PRIVATE bcrypt user32)
    set_target_properties(stellar_input_fixture PROPERTIES PREFIX "")
    # Beside clang-cl first: Visual Studio's ClangCL toolset ships clang there without putting it on PATH.
    get_filename_component(STELLAR_COMPILER_DIR ${CMAKE_CXX_COMPILER} DIRECTORY)
    find_program(CLANG_ASSEMBLER clang HINTS ${STELLAR_COMPILER_DIR} REQUIRED)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/fake_gameplay.obj
      COMMAND ${CLANG_ASSEMBLER} --target=x86_64-pc-windows-msvc -c ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/fake_gameplay.S -o ${CMAKE_CURRENT_BINARY_DIR}/fake_gameplay.obj
      DEPENDS fixtures/fake_gameplay.S VERBATIM)
    add_executable(stellar_raw_hook_tests tests/raw_hook_test.cpp src/frame_bus.cpp)
    target_include_directories(stellar_raw_hook_tests PRIVATE include)
    add_executable(stellar_frame_bus_fixture fixtures/frame_bus_fixture.cpp src/frame_bus.cpp)
    target_include_directories(stellar_frame_bus_fixture PRIVATE include)
    add_executable(stellar_pose_stream_fixture fixtures/pose_stream_fixture.cpp src/pose_stream.cpp)
    target_include_directories(stellar_pose_stream_fixture PRIVATE include)
    # Lua's side of the Windows sharing check (CI, beside BridgeRuntimeTest's contended run).
    add_executable(stellar_share_stress fixtures/share_stress.cpp)
    add_executable(stellar_input_fake_host fixtures/fake_host.cpp ${CMAKE_CURRENT_BINARY_DIR}/fake_gameplay.obj)
    target_link_options(stellar_input_fake_host PRIVATE /EXPORT:FakeGameplayReturn)
    target_link_libraries(stellar_input_fake_host PRIVATE winmm)
    # Imports the fixture as the game imports the adapter (by ordinal), then hot-patches its exports as Steam's overlay does.
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/fixture_ordinals.lib
      COMMAND ${CMAKE_AR} /nologo /machine:x64 /def:${CMAKE_CURRENT_SOURCE_DIR}/fixtures/fixture_ordinals.def /out:${CMAKE_CURRENT_BINARY_DIR}/fixture_ordinals.lib
      DEPENDS fixtures/fixture_ordinals.def VERBATIM)
    add_custom_target(stellar_fixture_ordinals DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/fixture_ordinals.lib)
    add_executable(stellar_input_overlay_host fixtures/overlay_host.cpp ${CMAKE_CURRENT_BINARY_DIR}/fake_gameplay.obj)
    add_dependencies(stellar_input_overlay_host stellar_fixture_ordinals stellar_input_fixture)
    target_link_libraries(stellar_input_overlay_host PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/fixture_ordinals.lib user32 winmm)
    target_link_options(stellar_input_overlay_host PRIVATE /EXPORT:FakeGameplayReturn)
  endif()
endif()
